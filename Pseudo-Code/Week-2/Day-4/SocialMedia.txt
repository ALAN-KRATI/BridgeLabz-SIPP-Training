GRAPH friendships = Map<User, List<User>>

FUNCTION recommendFriends(user)
    directFriends = friendships[user]
    queue = [user]
    visited = set with user
    distance = map with (user -> 0)
    suggestions = empty list

    WHILE queue not empty
        current = dequeue(queue)
        FOR each neighbor in friendships[current]
            IF neighbor not in visited
                visited.add(neighbor)
                distance[neighbor] = distance[current] + 1
                IF distance[neighbor] == 2
                    ADD neighbor to suggestions
                IF distance[neighbor] < 2
                    enqueue(neighbor)
    PRINT "Friend Suggestions: " + suggestions

MAIN
    DEFINE graph with Alice, Bob, Charlie, David, Emma, Frank
    CALL recommendFriends("Alice")
